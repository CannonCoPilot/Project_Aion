     # capitalised and the epithet lower-case, which is the convention every
     # journal follows and which keeps sentence-initial noise out.
     IN_TEXT = re.compile(r"\b([A-Z][a-z]{3,})\s+([a-z]{3,})\b")


     def load_lexicon(path: Path) -> dict[tuple[str, str], set[str]]:
         """(initial, epithet) -> {genus, ...} from a species reference list."""
         lex: dict[tuple[str, str], set[str]] = collections.defaultdict(set)
         if not path.exists():
             return lex
         import csv
         with path.open(newline="", encoding="utf-8", errors="replace") as fh:
             for row in csv.DictReader(fh):

                 m = BINOMIAL.match(sp[:1].upper() + sp[1:])
                 if m:
                     lex[(m.group(1)[0].upper(), m.group(2).lower())].add(m.group(1))
         return lex

         p = xml_dir / fname
         if not p.exists():
